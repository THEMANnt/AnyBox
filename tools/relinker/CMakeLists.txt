cmake_minimum_required(VERSION 3.25)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    project(AnyBoxRelinker VERSION 0.1.0 LANGUAGES C CXX)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/AnyBoxOptions.cmake")
    set(ANYBOX_RUNTIME_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
    option(BUILD_TESTING "Build relinker tests" OFF)
    include(CTest)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/Zydis.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/CompilerRuntime.cmake")
    anybox_add_zydis()
    if(BUILD_TESTING)
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../core/common" common)
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../core/relinker" core)
endif()

add_executable(anybox_relink Main.cpp)
target_link_libraries(anybox_relink PRIVATE anybox_relinker)
set_target_properties(anybox_relink PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ANYBOX_RUNTIME_DIRECTORY}")
anybox_add_compiler_runtime(anybox_relink)
